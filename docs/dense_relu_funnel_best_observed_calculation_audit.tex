\sectionbar{2. How each number is calculated}

\textbf{A. GEMM latency}

\begin{tabularx}{\linewidth}{@{}p{0.18\linewidth}Y@{}}
$C_{\mathrm{SCALE}}$ & Cycles returned by SCALE-Sim. \\
$W_{\mathrm{DRAM}}$ & One-byte words read from or written to DRAM. \\
$B_{\mathrm{DRAM}}$ & DRAM rate: $614$ words/cycle. \\
$t_{\mathrm{SA}}$ & SA cycle time: $1.00\ns$/cycle. \\
\end{tabularx}

\begin{align*}
C_{\mathrm{DRAM}} &= W_{\mathrm{DRAM}}/B_{\mathrm{DRAM}},\\
L_{\mathrm{array}} &= C_{\mathrm{SCALE}}\,t_{\mathrm{SA}},\\
L_{\mathrm{DRAM}} &= C_{\mathrm{DRAM}}\,t_{\mathrm{SA}},\\
L_{\mathrm{GEMM}} &= L_{\mathrm{array}}+L_{\mathrm{DRAM}}.
\end{align*}

Cycles are converted to nanoseconds before the two latency terms are added.
At $1.00$ GHz, one SA cycle happens to equal $1.00\ns$.

\textbf{GEMM dynamic energy}
\[
E_{\mathrm{GEMM,dyn}}
=W_{\mathrm{DRAM}}\times8\ \mathrm{bits/word}\times3.00\pj/\mathrm{bit}
+E_{\mathrm{SRAM}}.
\]
$E_{\mathrm{SRAM}}$ is the SRAM-energy value returned by CarbonPATH's SRAM
model for that GEMM.

\newpage
\textbf{B. FPGA ReLU}

Let $N_{\mathrm{elem}}$ be the number of tensor elements and
$N_{\mathrm{lanes}}$ the number processed in parallel:
\begin{align*}
N_{\mathrm{lanes}}
 &=\min\!\left(\left\lfloor\frac{10000\ \mathrm{CLBs}}
 {2\ \mathrm{CLBs/lane}}\right\rfloor,128\right)=128,\\
C_{\mathrm{ReLU}}
 &=\left\lceil N_{\mathrm{elem}}/N_{\mathrm{lanes}}\right\rceil,\\
t_{\mathrm{FPGA}}
 &=1/(400.00\ \mathrm{MHz})=2.50\ns/\mathrm{cycle},\\
L_{\mathrm{ReLU}}
 &=C_{\mathrm{ReLU}}\,t_{\mathrm{FPGA}},\\
E_{\mathrm{ReLU,dyn}}
 &=N_{\mathrm{elem}}\times0.10\pj/\mathrm{element}.
\end{align*}

\textbf{C. SA--FPGA movement}

Let $D$ be tensor bytes and $H$ be route hops.  Here $H=1$ for every crossing.
\begin{align*}
L_{\mathrm{move}}
 &=L_{\mathrm{setup}}+H L_{\mathrm{hop}}+D/B_{\mathrm{link}}\\
 &=2.00\ns+(1)(0.50\ns)+D/B_{\mathrm{link}},\\
E_{\mathrm{move}}
 &=D\times8\ \mathrm{bits/byte}\times0.25\pj/\mathrm{bit}.
\end{align*}
The first two latency terms are the saved transfer setup and one-hop delay.
The last term is serialization time: tensor bytes divided by link throughput.
Layer tables use the saved throughput before rounding.

\textbf{D. Complete layer}
\begin{align*}
L_i &=L_{i,\mathrm{operation}}+L_{i,\mathrm{move}},\\
E_i &=E_{i,\mathrm{baseline}}+E_{i,\mathrm{dynamic}}+E_{i,\mathrm{move}},\\
E_{i,\mathrm{baseline}}&=P_{\mathrm{system}}L_i.
\end{align*}
$P_{\mathrm{system}}$ is SA power plus FPGA power; its displayed value is
$4.48+2.00=6.48$ W.  Since W$\times$ns equals nJ, the last equation directly
produces nJ.

\sectionbar{6. What is and is not modeled}
\begin{itemize}
\item \textbf{Real GEMM path:} SCALE-Sim supplies the four GEMM cycle counts.
\item \textbf{Real analytical ReLU path:} \texttt{legacy\_fpga\_relu\_v1}; the
placeholder ReLU model is not selected.
\item \textbf{No Softmax:} a Softmax placeholder exists in the profile but is
never called because this graph has no Softmax layer.
\item \textbf{One byte per element:} every tensor is \texttt{int8}.
\item \textbf{Sequential accounting:} incoming movement is assigned to the
consumer and added to its layer time.
\item \textbf{Search limit:} this is the best observed one-SA/one-FPGA result,
not proof of a global optimum over unsupported multi-SA movement.
\end{itemize}
